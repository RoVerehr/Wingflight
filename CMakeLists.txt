cmake_minimum_required(VERSION 3.24)
project(Wingflight VERSION 0.3.13 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DEPENDENCY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Existing spdlog and rapidcsv source paths")
set(SPDLOG_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPDLOG_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${DEPENDENCY_ROOT}/vendor/spdlog" dependencies/spdlog)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_XAUDIO_WIN8 OFF CACHE BOOL "" FORCE)
set(BUILD_XAUDIO_WIN10 OFF CACHE BOOL "" FORCE)
set(BUILD_XAUDIO_REDIST OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/DirectXTK)
add_library(Microsoft::DirectXTK ALIAS DirectXTK)
set(spdlog_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cmake/dependency-config" CACHE PATH "" FORCE)
set(directxtk_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cmake/dependency-config" CACHE PATH "" FORCE)
set(RAPIDCSV_INCLUDE_DIRS "${DEPENDENCY_ROOT}/vendor/rapidcsv/src" CACHE PATH "" FORCE)
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_HDE64 "${CMAKE_CURRENT_SOURCE_DIR}/vendor/minhook" CACHE PATH "" FORCE)
add_subdirectory(vendor/CommonLibSSE-NG)
include(FetchContent)
FetchContent_GetProperties(hde64 SOURCE_DIR WINGFLIGHT_MINHOOK_SOURCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${WINGFLIGHT_MINHOOK_SOURCE}" dependencies/minhook)
add_library(CommonLib ALIAS CommonLibSSE)
add_library(Wingflight SHARED src/plugin.cpp src/config.cpp src/wings.cpp src/wing_runtime.cpp src/glide_clip.cpp src/lower_body.cpp src/armed_sprint_clip.cpp src/awu_compatibility.cpp src/settings_menu.cpp src/settings_file.cpp src/runtime_support.cpp src/native_contract.cpp src/flight_audio.cpp src/diagnostics.cpp src/vampire_runtime.cpp)
target_include_directories(Wingflight PRIVATE vendor/SKSE-Menu-Framework-3-API)
target_include_directories(Wingflight PRIVATE "${WINGFLIGHT_MINHOOK_SOURCE}/src/hde")
target_link_libraries(Wingflight PRIVATE CommonLib minhook)
target_compile_definitions(Wingflight PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_precompile_headers(Wingflight PRIVATE <RE/Skyrim.h> <SKSE/SKSE.h>)
if(MSVC)
  target_compile_options(Wingflight PRIVATE /W4 /permissive- /wd4200 /wd4201 /Zi)
  target_link_options(Wingflight PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF "/MAP:${CMAKE_BINARY_DIR}/Wingflight.map")
endif()
